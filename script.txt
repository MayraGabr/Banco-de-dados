-- Por cpf aparece as informações do cliente e todas as compras com informações da venda:

SELECT
    c.*,         
    v.codigo_venda,
    v.data_venda,
    v.valor_total,
    v.cpf_funcionario,
    vp.*          
FROM Cliente c
LEFT JOIN Venda v ON v.cpf_cliente = c.cpf
LEFT JOIN Venda_has_Produto vp ON vp.codigo_venda = v.codigo_venda
WHERE c.cpf = '11111111111';
